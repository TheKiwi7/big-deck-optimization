# Co-op without action races (E1, host): mod on, plays 4 cards per turn. Only one player acts at a time, so the action order
# can't differ between peers: any divergence would come from a different computation. See coopsolo1-client.txt.
timeout 1800
mplobby NECROBINDER 2
waitfor run 240
cmd bdperf checks off
wait 8
cmd bigdeck import ref all all {hash:ref}
cmd bigdeck import ref all cards {hash:ref}
cmd bigdeck import ref all cards {hash:ref}
cmd bigdeck import ref all cards {hash:ref}
enemyhp 999999
wait 25
cmd fight CULTISTS_NORMAL
waitfor idle 1200
state t1
play 4
endturn
waitfor idle 1200
state t2
play 4
endturn
waitfor idle 1200
state t3
cmd kill all
waitfor nocombat 900
wait 20
quit
