# Co-op without action races (E2, client): mod on, plays 4 cards per turn. Only one player acts at a time, so the action order
# can't differ between peers: any divergence would come from a different computation. See coopsolo2-host.txt.
timeout 1800
mplobby IRONCLAD 2
waitfor run 240
cmd bdperf checks off
wait 14
cmd bigdeck import ref all cards {hash:ref}
cmd bigdeck import ref all cards {hash:ref}
cmd bigdeck import ref all cards {hash:ref}
cmd bigdeck import ref all cards {hash:ref}
enemyhp 999999
waitfor combat 1200
waitfor idle 1200
state t1
play 4
endturn
waitfor idle 1200
state t2
play 4
endturn
waitfor idle 1200
state t3
waitfor nocombat 1200
wait 25
quit
