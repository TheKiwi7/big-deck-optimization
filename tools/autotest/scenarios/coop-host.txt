# Local co-op (host), see coop.ps1: ~3,750 cards per player (+ the reference relics on the host), two combats,
# turn states logged on both sides to compare. Every machine needs the same bigdeck/ref.run.
timeout 2400
mplobby NECROBINDER 2
waitfor run 240
cmd bdperf checks off
wait 8
cmd bigdeck import ref all all {hash:ref}
cmd bigdeck import ref all cards {hash:ref}
cmd bigdeck import ref all cards {hash:ref}
cmd bigdeck import ref all cards {hash:ref}
enemyhp 999999
wait 25
cmd fight CULTISTS_NORMAL
waitfor idle 900
cmd bdperf combatstart
state t1
play 4
endturn
waitfor idle 900
state t2
play 4
endturn
waitfor idle 900
state t3
cmd bdperf verify
cmd kill all
waitfor nocombat 600
wait 8
cmd fight CULTISTS_NORMAL
waitfor idle 900
state f2t1
play 3
endturn
waitfor idle 900
state f2t2
cmd bdperf verify
cmd kill all
waitfor nocombat 600
wait 20
quit
